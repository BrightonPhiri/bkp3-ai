Python File Cleaner

Here's a Python script that deletes files under 5KB:

```python import os from pathlib import Path def delete_small_files(directory, size_limit_kb=5, recursive=False): """Deletes all files smaller than size_limit_kb kilobytes in the given directory. :param directory: Path to the directory to scan. :param size_limit_kb: Size limit in kilobytes. Files smaller than this will be deleted. :param recursive: If True, scan subdirectories recursively. """ size_limit_bytes = size_limit_kb * 1024 # Walk through directory tree if recursive if recursive: for root, dirs, files in os.walk(directory): for filename in files: filepath = os.path.join(root, filename) try: if os.path.getsize(filepath) < size_limit_bytes: print(f"Deleting {filepath} ({os.path.getsize(filepath)} bytes)") os.remove(filepath) except Exception as e: print(f"Error deleting {filepath}: {e}") else: # Only scan the specified directory for filename in os.listdir(directory): filepath = os.path.join(directory, filename) if os.path.isfile(filepath): try: if os.path.getsize(filepath) < size_limit_bytes: print(f"Deleting {filepath} ({os.path.getsize(filepath)} bytes)") os.remove(filepath) except Exception as e: print(f"Error deleting {filepath}: {e}") if __name__ == "__main__": directory_to_clean = input("Enter the directory path to clean: ").strip() recursive_choice = input("Delete files in subdirectories as well? (y/n): ") recursive = recursive_choice.lower() == "y" delete_small_files(directory_to_clean, 5, recursive=recursive) ```
